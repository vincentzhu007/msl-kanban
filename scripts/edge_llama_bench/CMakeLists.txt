cmake_minimum_required(VERSION 3.16)
project(edge_llama_bench C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_CPP_DIR "" CACHE PATH "Path to the pinned llama.cpp checkout")
if(NOT EXISTS "${LLAMA_CPP_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set LLAMA_CPP_DIR to a llama.cpp checkout")
endif()
add_subdirectory("${LLAMA_CPP_DIR}" llama)
add_executable(edge-llama bench.cpp)
target_link_libraries(edge-llama PRIVATE llama)
target_link_options(edge-llama PRIVATE -static-libstdc++)

add_executable(recipe-llama bench_recipe.cpp)
target_link_libraries(recipe-llama PRIVATE llama)
target_link_options(recipe-llama PRIVATE -static-libstdc++)
